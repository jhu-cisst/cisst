cmake_minimum_required (VERSION 3.16)
project (cisstCppUnitConsumer LANGUAGES NONE)

list (PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../../../cmake")

# Isolate all searches from CppUnit installations on the host.
set (_root "${CMAKE_CURRENT_BINARY_DIR}/root")
set (CMAKE_FIND_ROOT_PATH "${_root}")
set (CMAKE_FIND_ROOT_PATH_MODE_PACKAGE ONLY)
set (CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
set (CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set (CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
set (CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY FALSE)
set (CMAKE_PREFIX_PATH "${_root}")
set (CppUnit_DIR "${_root}/share/CppUnit")

if (NOT CASE STREQUAL "missing" AND NOT CASE STREQUAL "required")
  file (MAKE_DIRECTORY "${_root}/include/cppunit" "${_root}/lib")
  file (WRITE "${_root}/include/cppunit/TestCase.h" "")
  file (WRITE "${_root}/lib/${CMAKE_STATIC_LIBRARY_PREFIX}cppunit${CMAKE_STATIC_LIBRARY_SUFFIX}" "")
  if (NOT CASE STREQUAL "release-only")
    file (WRITE "${_root}/lib/${CMAKE_STATIC_LIBRARY_PREFIX}cppunitd${CMAKE_STATIC_LIBRARY_SUFFIX}" "")
  endif ()
endif ()

if (CASE STREQUAL "config" OR CASE STREQUAL "namespaced-config")
  set (_target CppUnit)
  if (CASE STREQUAL "namespaced-config")
    set (_target CppUnit::CppUnit)
  endif ()
  # Like vcpkg, this config intentionally has no guard around add_library.
  file (WRITE "${CppUnit_DIR}/CppUnitConfig.cmake"
        "add_library (${_target} INTERFACE IMPORTED)\nset (CppUnit_FOUND TRUE)\n")
  # A config must take priority over a failing manual search.
  set (CPPUNIT_INCLUDE_DIR CPPUNIT_INCLUDE_DIR-NOTFOUND)
  set (CPPUNIT_LIBRARY CPPUNIT_LIBRARY-NOTFOUND)
elseif (CASE STREQUAL "existing")
  add_library (CppUnit INTERFACE IMPORTED)
endif ()

if (CASE STREQUAL "required")
  find_package (CppUnit MODULE REQUIRED)
  message (FATAL_ERROR "REQUIRED discovery unexpectedly returned")
endif ()

# Exercise the same result-variable check used by the installed cisst package.
file (WRITE "${CMAKE_CURRENT_BINARY_DIR}/dependency.cmake"
      "include (CMakeFindDependencyMacro)\nfind_dependency (CppUnit MODULE)\n")
set (CMAKE_FIND_PACKAGE_NAME Consumer)
set (Consumer_FOUND TRUE)
set (Consumer_FIND_QUIETLY TRUE)
include ("${CMAKE_CURRENT_BINARY_DIR}/dependency.cmake")
if (CASE STREQUAL "missing")
  if (Consumer_FOUND OR CppUnit_FOUND OR CPPUNIT_FOUND OR TARGET CppUnit::CppUnit)
    message (FATAL_ERROR "Missing CppUnit was reported as found")
  endif ()
  return ()
endif ()
if (NOT Consumer_FOUND OR NOT CppUnit_FOUND OR NOT CPPUNIT_FOUND OR NOT TARGET CppUnit::CppUnit)
  message (FATAL_ERROR "CppUnit discovery failed: ${Consumer_NOT_FOUND_MESSAGE}")
endif ()
if (CASE STREQUAL "config" OR CASE STREQUAL "namespaced-config")
  get_target_property (_type ${_target} TYPE)
  if (NOT _type STREQUAL "INTERFACE_LIBRARY")
    message (FATAL_ERROR "Manual discovery took priority over the package config")
  endif ()
elseif (CASE STREQUAL "manual" AND WIN32)
  get_target_property (_debug CppUnit::CppUnit IMPORTED_LOCATION_DEBUG)
  if (NOT _debug MATCHES "cppunitd")
    message (FATAL_ERROR "Manual discovery did not select the separate debug library")
  endif ()
endif ()
if (CASE STREQUAL "release-only")
  get_target_property (_debug CppUnit::CppUnit IMPORTED_LOCATION_DEBUG)
  if (NOT _debug STREQUAL CPPUNIT_LIBRARY)
    message (FATAL_ERROR "Release-only CppUnit did not supply a debug fallback")
  endif ()
endif ()

# Bypass find_dependency's cache to verify the finder itself is repeatable.
find_package (CppUnit MODULE REQUIRED)
find_package (CppUnit MODULE REQUIRED)
